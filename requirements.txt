# Pinned deliberately. This pipeline runs unattended from cron every morning,
# so an upstream point release changing a default would land in production
# without anyone choosing it. To upgrade: bump a version here, run the tests,
# then re-run the full pipeline and check the row count is unchanged.
#
# Verified working together on 2026-08-21 against the full 36-year corpus.

pandas==3.0.5
plotly==6.9.0

# NOT numpy 2.x, deliberately - do not "upgrade" this without reading on.
#
# The numpy 2.x Linux wheels are compiled with an x86-64-v2 baseline (SSE4.2,
# POPCNT). The production cPanel box does not have it, so merely importing
# pandas dies with:
#
#   RuntimeError: NumPy was built with baseline optimizations: (X86_V2)
#   but your machine doesn't support: (X86_V2).
#
# Note it installs perfectly happily and only fails at import, so a green
# `pip install` proves nothing here.
#
# 1.26.4 is the newest release that still ships a wheel that machine can run,
# and pandas 3.0.5 accepts numpy>=1.26.0 on Python 3.12. Verified on this
# pairing: 104/104 tests pass and the full corpus reads correctly.
numpy==1.26.4

# Not imported by our code, but pandas uses it automatically for string columns
# when it is installed. On the full 36-year corpus that cut peak memory from
# 11.0 GB to 6.3 GB with no code change at all.
#
# Optional - pandas 3.0.5 does not require it, and everything works without it
# (just hungrier). If the server hits the same x86-64-v2 wall described above,
# uninstalling pyarrow there is a legitimate fix; check the memory headroom on
# a full run if you do, because that 11 GB figure is the one you get back.
pyarrow==25.0.1

# analysis.ipynb only - not needed to run the pipeline. plotly's fig.show()
# refuses to render inside a notebook without it ("Mime type rendering requires
# nbformat>=4.2.0"), and nothing else in the chain pulls it in.
nbformat==5.11.1

# Test suite only - not needed to run the pipeline.
pytest==9.1.1

# Deliberately absent, so nobody adds them back on a hunch:
#   scipy      - the notebook finds outliers with an IQR fence on log prices,
#                which is plain numpy. Nothing here needs scipy.
#   matplotlib - the notebook charts with plotly. Nothing imports matplotlib.
